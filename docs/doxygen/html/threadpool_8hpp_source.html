32inline ThreadPool::ThreadPool(size_t threads) {
33 workers.reserve(threads);
34 for (size_t i = 0; i < threads; ++i) {
35 workers.emplace_back([this] {
36 while (true) {
37 std::function<void()> task;
38 {
39 std::unique_lock<std::mutex> lock(queue_mutex);
40 condition.wait(lock, [this] {
41 return stop || !tasks.empty();
42 });
43 if (stop && tasks.empty())
44 return;
45 task = std::move(tasks.front());
46 tasks.pop();
47 }
48 task();
49 }
50 });
51 }
52}

54template<class F, class... Args>
55auto ThreadPool::enqueue(F&& f, Args&&... args)
56 -> std::future<std::invoke_result_t<F, Args...>> {
57 using return_type = std::invoke_result_t<F, Args...>;
58
59 // wrap the callable+args into a packaged_task
60 auto task_ptr = std::make_shared<std::packaged_task<return_type()>>(
61 [fn = std::forward<F>(f), ... a = std::forward<Args>(args)]() mutable {
62 return fn(a...);
63 }
64 );
65
66 std::future<return_type> res = task_ptr->get_future();
67 {
68 std::lock_guard<std::mutex> lock(queue_mutex);
69 if (stop)
70 throw std::runtime_error("enqueue on stopped ThreadPool");
71 tasks.emplace([task_ptr]{ (*task_ptr)(); });
72 }
73 condition.notify_one();
74 return res;
75}
