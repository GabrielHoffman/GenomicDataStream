41 DelayedStream( const shared_ptr<tatami::NumericMatrix> &ptr, const vector<string> &rowNames, const int &chunkSize)
42 : GenomicDataStream(), ptr(ptr), rowNames(rowNames), chunkSize(chunkSize) {
43
44 if( chunkSize < 1){

185 void getNextChunk( DataChunk<Eigen::MatrixXd> & chunk, int start, int len){
186
187 int length = min(len, NR - start);
188
189 // get workspace as dense row
190 auto wrk = ptr->dense_row();
191
192 // loop through rows
193 for (int i = 0; i < length; i++) {
194 // get data for row start + i
195 auto extracted = wrk->fetch(start + i, buffer.data());
196
197 // copy data into output vector in column i
198 memcpy(output.data() + NC*i, extracted, NC*sizeof(double));
199 }
200
201 // get feature names
202 mInfo->setRowNames(rowNames, start, start + length);
203
204 Eigen::MatrixXd M = Eigen::Map<Eigen::MatrixXd>(output.data(), NC, length);
205
206 chunk = DataChunk<Eigen::MatrixXd>( M, mInfo );
207 }

220 // original code based on stand-alone beachmat
221 bool getNextChunk_helper(){
222
223 // if end of file reached, return false
224 if( ! continueIterating ) return continueIterating;
225
226 // get workspace as dense row
227 auto wrk = ptr->dense_row();
228
229 // if remaning rows is less than chunkSize,
230 // then set chunkSize to stop at end
231 chunkSize = min(chunkSize, NR - pos);
232
233 // loop through rows
234 for (int i = 0; i < chunkSize; i++) {
235 // get data for row pos + i
236 auto extracted = wrk->fetch(pos + i, buffer.data());
237
238 // copy data into output vector in column i
239 memcpy(output.data() + NC*i, extracted, NC*sizeof(double));
240 }
241
242 // get feature names
243 mInfo->setRowNames(rowNames, pos, pos+chunkSize);
244
245 // increment current position
246 pos += chunkSize;
247
248 // if current position is less than number of rows
249 // return true to continue and get text chunk
250 continueIterating = (pos < NR);
251
252 return true;
253 }
254};
