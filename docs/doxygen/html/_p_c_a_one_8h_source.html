37PCA pcaone( const shared_ptr<GenomicDataStream> gds,
38 const string &region,
39 int m,
40 int k,
41 int nchunks,
42 int s = 20,
43 int p = 7,
44 int B = 64,
45 int threads = 4,
46 const bool verbose = true,
47 const bool scaleAndCenter = true,
48 ostream& outstrm = Rcpp::Rcout) {
49
50 auto regions = splitRegionString( region );
51
52 int n = gds->n_samples();
53
54 const int l = k + s;
55 auto randomEngine = std::default_random_engine{};
56 Eigen::MatrixXd Omg = StandardNormalRandom<Eigen::MatrixXd, std::default_random_engine>(n, l, randomEngine);
57 Eigen::MatrixXd Omg2 = Omg;
58 Eigen::MatrixXd H1 = Eigen::MatrixXd::Zero(n, l);
59 Eigen::MatrixXd H2 = Eigen::MatrixXd::Zero(n, l);
60 Eigen::MatrixXd H(n, l), G(m, l), R(l, l), Rt(l, l);
61
62 // Parallel part using Thread Building Blocks
63 GenomicDataStreamParallel<Eigen::MatrixXd> gsp(gds->getParam(), regions, nchunks, threads);
64
65 size_t band = ceil((double) nchunks / B);
66 vector<string> featureIds;
67 std::mutex pcaMutex;
68
69 for (int pi = 0; pi <= p; pi++) {
70
71 if( verbose ){
72 outstrm << "\rEpoch " << pi << " / " << p << " ";
73 }
74
75 if (std::pow(2, pi) >= B) {
76 // reset H1, H2 to zero
77 H1.setZero();
78 H2.setZero();
79 }
80 band = std::fmin(band * 2, nchunks);
81
82 size_t i{1}, start{0};
83 gsp.processChunks([&](const gds::DataChunk<Eigen::MatrixXd> &chunk, size_t b) {
84 auto Ab = chunk.getData();
85
86 if( scaleAndCenter ){
87 standardize(Ab);
88 Ab /= sqrt(n-1);
89 }
90 {
91 std::lock_guard<std::mutex> lock(pcaMutex);
92
93 if( pi == 0){
94 auto tmp = chunk.getInfo<VariantInfo>()->getFeatureNames();
95 featureIds.insert(featureIds.end(),
96 tmp.begin(), tmp.end());
97 }
98
99 G.middleRows(start, Ab.cols()).noalias() = Ab.transpose() * Omg;
100 if (i <= band / 2)
101 H1.noalias() += Ab * G.middleRows(start, Ab.cols());
102 else
103 H2.noalias() += Ab * G.middleRows(start, Ab.cols());
104 bool adjacent =
105 (pi > 0 && (b + 1) == std::pow(2, pi - 1) && std::pow(2, pi) < B);
106 if((!((b + 1) < band && !adjacent)) && ((i == band) || (i == band / 2) || adjacent)){
107 H = H1 + H2;
108 Eigen::HouseholderQR<Eigen::MatrixXd> qr(H);
109 Omg.noalias() = qr.householderQ() * Eigen::MatrixXd::Identity(n, l);
110 flipOmg(Omg2, Omg);
111 if (i == band) {
112 H1.setZero();
113 i = 0;
114 } else {
115 H2.setZero();
116 }
117 }
118 start += Ab.cols();
119 i++;
120 }
121 }); // end processChunks
122 } // end for
123
124 // get USV
125 if( verbose ){
126 outstrm << "\rFinal decompositions";
127 }
128 {
129 Eigen::HouseholderQR<Eigen::Ref<Eigen::MatrixXd>> qr(G);
130 R.noalias() = Eigen::MatrixXd::Identity(l, m) * qr.matrixQR().triangularView<Eigen::Upper>();
131 G.noalias() = qr.householderQ() * Eigen::MatrixXd::Identity(m, l);
132 }
133 {
134 Eigen::HouseholderQR<Eigen::Ref<Eigen::MatrixXd>> qr(G);
135 Rt.noalias() = Eigen::MatrixXd::Identity(l, m) * qr.matrixQR().triangularView<Eigen::Upper>();
136 G.noalias() = qr.householderQ() * Eigen::MatrixXd::Identity(m, l);
137 }
138
139 R = Rt * R;
140
141 Eigen::MatrixXd out = R.transpose().fullPivHouseholderQr().solve(H.transpose());
142 Eigen::JacobiSVD<Eigen::MatrixXd> svd(out, Eigen::ComputeThinU | Eigen::ComputeThinV);
143
144 if( verbose ){
145 outstrm << "\rCompleted " << std::endl;
146 }
147
148 return PCA(svd.matrixV().leftCols(k),
149 G * svd.matrixU().leftCols(k),
150 svd.singularValues().head(k),
151 featureIds);
152}
