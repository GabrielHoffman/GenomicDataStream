73template<typename MatrixType>
74void flipOmg(MatrixType & Omg2, MatrixType & Omg)
75{
76 for(Eigen::Index i = 0; i < Omg.cols(); ++i)
77 {
78 // if signs of half of values are flipped then correct signs.
79 if((Omg2.col(i) - Omg.col(i)).array().abs().sum() > 2 * (Omg2.col(i) + Omg.col(i)).array().abs().sum())
80 {
81 Omg.col(i) *= -1;
82 }
83 }
84 Omg2 = Omg;

87template<typename MatrixType>
88Eigen::Matrix<typename MatrixType::Scalar, Eigen::Dynamic, Eigen::Dynamic> center_and_scale(
89 const MatrixType & matrix,
90 const Eigen::Ref<const Eigen::VectorXd> & center,
91 const Eigen::Ref<const Eigen::VectorXd> & scale,
92 const bool byrow)
93{
94 using Scalar = typename MatrixType::Scalar;
95 assert(center.size() == scale.size() && "the size of center and scale are different");
96
97 // Convert input matrix to dense (handles both sparse and dense matrices)
98 Eigen::Matrix<Scalar, Eigen::Dynamic, Eigen::Dynamic> mat = matrix;
99 if(byrow)
100 {
101 mat.colwise() -= center;
102 // Scale rows
103 mat = scale.asDiagonal() * mat;
104 }
105 else
106 {
107 mat.rowwise() -= center.transpose();
108 // Scale cols
109 mat = mat * scale.asDiagonal();
110 }
111
112 return mat;
113}
